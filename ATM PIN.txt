int readPinCode()
{

}

bool readUntilCorrect()
{
    int code = readPinCode();
    while (code != 1234)
    {
        cout << "Wrong PIN" << endl;
        system("color 4F");
        code = readPinCode();
    }
    return true;
}

void displayBalance()
{

        cout << "*******************************"
             << endl

             << endl;
    }
}
