float calculateAvg(int n1, int n2, int n3)
{
    return (n1 + n2 + n3) / 3.0;
}

StudentStatus checkMark(float mark)
{
    return (mark >= 50) ? StudentStatus::Pass : StudentStatus::Failed;
}

void printAvgRes(float average)
{
    cout << "***********************************\n\n";
    cout << "Your average is: " << average << endl;
    if (checkMark(average) == StudentStatus::Pass)
        cout << "Passed :)" << endl;
    else
        cout << "Failed :( ." << endl;
}
