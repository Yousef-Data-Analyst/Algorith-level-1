#include <iostream>
#include <cmath>
using namespace std;

int readPositive(const string &message)
{
    int num = -1;
    do
    {
        cout << message;
        cin >> num;
    } while (num < 0);
    return num;
}

unsigned long long calculateFactorial(int num)
{
    unsigned long long factorial = 1;
    for (int i = 2; i <= num; i++)
        factorial *= i;
    return factorial;
}

void printFactorial(int num)
{
    cout << "***********************************\n";
    cout << "Factorial of " << num << " is " << calculateFactorial(num) << endl
         << endl;
}
int main()
{

    printFactorial(readPositive("Enter positive number to get the Factoial: "));
    return 0;
}
