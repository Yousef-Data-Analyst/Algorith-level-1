#include <iostream>
#include <cmath>
using namespace std;

void readDiameter(float &diameter)
{
    cout << "Enter the Diameter: ";
    cin >> diameter;
}

float calculateCircleArea(float diameter)
{
    return (3.14 * pow(diameter, 2)) / 4.0;
}

void printResult(float res)
{
    cout << "*********************************\n";
    cout << "Circle Area: " << res << endl;
}
int main()
{
    float diameter;

    readDiameter(diameter);
    printResult(calculateCircleArea(diameter));
    return 0;
}
