#include <iostream>
#include <cmath>
using namespace std;

float readDiameter()
{
    float diameter;
    cout << "Enter the square length: ";
    cin >> diameter;
    return diameter;
}

float calculateCircleArea(float diameter)
{
    return (3.14 * pow(diameter, 2)) / 4.0;
}

void printResult(float res)
{
    cout << "*********************************\n";
    cout << "Circle Area: " << res << endl;
}
int main()
{

    printResult(calculateCircleArea(readDiameter()));
    return 0;
}
