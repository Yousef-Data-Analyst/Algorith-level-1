#include<iostream>
using namespace std;

UserInfo readUserInfo()
{
    UserInfo info;
    cout << "Enter your age : ";
    cin >> info.age;
    cout << "Do you have a driver license? (0 = No, 1 = Yes): ";
    cin >> info.HasDriverLicense;
    
    return info;
}

bool isAccepted(const UserInfo &info)
{
    return (info.age > 21 && info.HasDriverLicense);
}

void HiredOrNot(const UserInfo &info)
{
    if (isAccepted(info))
        cout << "Hired" << endl;
    else
        cout << "Rejected" << endl;
}

int main()
{
    HiredOrNot(readUserInfo());

    return 0;
}
