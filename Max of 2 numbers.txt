#include<iostream>
using namespace std;

int readIntNumber(const string &message)
{
    cout << message;
    int number;
    cin >> number;
    return number;
}

void read2Numbers(int &num1, int &num2)
{
    num1 = readIntNumber("Enter the first number: ");
    num2 = readIntNumber("Enter the second number: ");
}

int maxOf2Numbers(int num1, int num2)
{
    if (num1 > num2)
        return num1;
    else
        return num2;
}

void printMaxOf2Numbers(int num1, int num2)
{
    cout << "************************************" << endl
         << endl;
    cout << "Maximum: " << maxOf2Numbers(num1, num2) << endl;
}

int main()
{
    int num1 , num2;
    read2Numbers(num1,num2);
    printMaxOf2Numbers(num1,num2);
