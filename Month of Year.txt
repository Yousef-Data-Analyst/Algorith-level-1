#include <iostream>
#include <cmath>
#include <string>
#include <iomanip>
using namespace std;

enum Months
{
    Jan = 1,
    Feb = 2,
    Mar = 3,
    Apr = 4,
    May = 5,
    Jun = 6,
    Jul = 7,
    Aug = 8,
    Sep = 9,
    Oct = 10,
    Nov = 11,
    Dec = 12
};

Months readMonthBetween(int from, int to)
{
    int month;
    do
    {
        cout << "Enter Month number 1 to 12: ";
        cin >> month;
    } while (!(month <= to && month >= from));
    return Months(month);
}

string getMonth(Months month)
{
    switch (month)
    {
    case Months::Jan:
        return "January";
    case Months::Feb:
        return "February";
    case Months::Mar:
        return "March";
    case Months::Apr:
        return "April";
    case Months::May:
        return "May";
    case Months::Jun:
        return "June";
    case Months::Jul:
        return "July";
    case Months::Aug:
        return "August";
    case Months::Oct:
        return "October";
    case Months::Sep:
        return "September";
    case Months::Dec:
        return "December";
    case Months::Nov:
        return "November";
    default:
        return "Invalid";
    }
}

void printMonth(Months month)
{
    cout << "*****************************" << endl
         << endl;
    cout << getMonth(month) << " (Month " << month << ")" << endl;
}

int main()
{
    printMonth(readMonthBetween(1 , 12));
    return 0;
}
