#include <iostream>
#include <cmath>
using namespace std;

int readNumber()
{
    int number;
    cout << "Enter a number: ";
    cin >> number;
    return number;
}

void printNumbersFrom1_to_N_UsingDoWhile(int n)
{
    int counter = 1;
    do
    {
        cout << counter << endl;
        counter++;
    } while (counter <= n);
}

void printNumbersFrom1_to_N_UsingWhile(int n)
{
    int counter = 0;
    while (counter < n)
    {
        cout << ++counter << endl;
    }
}

void printNumberFrom1_to_N_UsingFor(int n)
{
    for (int i = 1; i <= n; i++)
    {
        cout << i << endl;
    }
}

int main()
{
    int n = readNumber();
    printNumbersFrom1_to_N_UsingDoWhile(n);
    cout << "\n******************************\n";
    printNumbersFrom1_to_N_UsingWhile(n);
    cout << "\n********************************\n";
    printNumberFrom1_to_N_UsingFor(n);
    return 0;
}
